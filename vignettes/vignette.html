# Count data 
#counts(spe)
# Assign the readable gene names to the rows
rownames(spe) <- rowData(spe)$gene_name

# Calculate variance across all spots for each gene
rv <- rowVars(counts(spe))
names(rv) <- rownames(spe)

# Order the entire object by raw count variance (highest to lowest)
spe_ordered <- spe[order(rv, decreasing = TRUE), ]

# Subset to the top 200 genes
spe_vignette <- spe_ordered[1:200, ]
# Key markers for norepinephrine neurons
lc_markers <- c("TH", "DBH", "SLC6A2", "GCH1", "DDC")
# Check which ones are in your high-variance vignette set
present_markers <- intersect(lc_markers, rownames(spe_vignette))
